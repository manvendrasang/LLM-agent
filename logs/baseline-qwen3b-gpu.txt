import time

def exponential_backoff_with_retries(
    func: callable,
    max_retries: int = 5,
    initial_delay: float = 1.0,
    backoff_factor: float = 2.0,
    jitter: float = 0.5
) -> callable:
    """
    Decorator to implement exponential backoff with a maximum number of retries.

    :param func: The function to be decorated.
    :param max_retries: The maximum number of retries.
    :param initial_delay: The initial delay in seconds.
    :param backoff_factor: The factor by which the delay is multiplied after each retry.
    :param jitter: The amount of random jitter to add to the delay.
    :return: The decorated function.
    """
    def wrapper(*args, **kwargs):
        retries = 0
        while retries < max_retries:
            try:
                return func(*args, **kwargs)
            except Exception as e:
                retries += 1
                delay = initial_delay * (backoff_factor ** retries) + jitter * (2 * random.random() - 1)
                print(f"Attempt {retries} failed. Retrying in {delay:.2f} seconds...")
                time.sleep(delay)
        raise Exception(f"Failed to execute {func.__name__} after {max_retries} retries.")
    return wrapper
